% ── 必需宏包 ──
\usepackage{amsmath}    % align 等数学环境
\usepackage{etoolbox}   % 条件判断

% ── 表格跨页 ──
\usepackage{longtable}
\usepackage{booktabs}
\usepackage{array}
\usepackage{xltabular}  % longtable + tabularx 合体

% 行距
\renewcommand{\arraystretch}{1.3}
\setlength{\LTpre}{0pt}
\setlength{\LTpost}{0pt}

% ── 代码块跨页 (tcolorbox 接管 Shaded 环境) ──
\usepackage{tcolorbox}
\tcbuselibrary{breakable, skins, listings}

% 确保 Shaded 环境存在（不同 Pandoc/Eisvogel 版本行为不同）
\makeatletter
\@ifundefined{Shaded}{\newenvironment{Shaded}{}{}}{}
\makeatother

% 用 tcolorbox 接管，获得 breakable 能力
\renewenvironment{Shaded}{
  \begin{tcolorbox}[
    breakable,
    enhanced,
    colback=gray!3,
    colframe=gray!30,
    boxrule=0.5pt,
    arc=2pt,
    left=6pt,
    right=6pt,
    top=6pt,
    bottom=6pt,
    before skip=8pt,
    after skip=8pt
  ]
}{
  \end{tcolorbox}
}

% ── listings 基础配置 (如果使用 --listings) ──
\lstset{
  basicstyle=\ttfamily\small,
  breaklines=true,
  breakatwhitespace=false,
  keepspaces=true,
  extendedchars=true,
  showstringspaces=false,
  frame=none,
  numbers=none
}

% ── 防止 Overfull hbox ──
\emergencystretch=1em
\usepackage{microtype}

% ── 浮动体容许度 ──
\renewcommand{\floatpagefraction}{0.85}
\renewcommand{\textfraction}{0.1}
\renewcommand{\topfraction}{0.9}
\setcounter{totalnumber}{5}

% ── 孤儿/寡行控制 ──
\widowpenalty=10000
\clubpenalty=10000
